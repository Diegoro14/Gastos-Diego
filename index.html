function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var data = JSON.parse(e.postData.contents);

  // Editar o eliminar una fila existente
  if (data.action === 'update' || data.action === 'delete') {
    var row = parseInt(data.id);
    var ok = row >= 2 && row <= sheet.getLastRow();
    if (ok) {
      var cur = sheet.getRange(row, 1, 1, 7).getValues()[0];
      ok = String(cur[2]) === String(data.expectedDesc) && parseInt(cur[5]) === parseInt(data.expectedAmount);
    }
    if (!ok) {
      return ContentService.createTextOutput(JSON.stringify({ result: "error", message: "La fila no coincide" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    if (data.action === 'delete') {
      sheet.deleteRow(row);
    } else {
      var r = sheet.getRange(row, 1, 1, 7);
      r.setNumberFormats([["@", "@", "@", "@", "@", "0", "@"]]);
      r.setValues([[data.date, data.time, data.description, data.category, data.method,
        parseInt(String(data.amount).replace(/\D/g, '')) || 0, data.status]]);
    }
    try { actualizarResumen(); } catch (err) {}
    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Limpiar monto a número entero ($ CLP)
  var amountClean = parseInt(String(data.amount).replace(/\D/g, '')) || 0;

  var tz = Session.getScriptTimeZone();
  var now = new Date();
  var newRow = [
    data.date || Utilities.formatDate(now, tz, "yyyy-MM-dd"),
    data.time || Utilities.formatDate(now, tz, "HH:mm"),
    data.description || data.desc || "Gasto",
    data.category || "Compras Varias",
    data.method || "Débito - Apple Pay",
    amountClean,
    data.status || "Pagado"
  ];

  // Fecha y hora como texto para que Sheets no las convierta
  var range = sheet.getRange(sheet.getLastRow() + 1, 1, 1, 7);
  range.setNumberFormats([["@", "@", "@", "@", "@", "0", "@"]]);
  range.setValues([newRow]);

  try { actualizarResumen(); } catch (err) {}
  return ContentService.createTextOutput(JSON.stringify({ result: "success", row: newRow }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var rows = sheet.getDataRange().getValues();

  // Ignorar encabezado (fila 1)
  var expenses = [];
  for (var i = 1; i < rows.length; i++) {
    expenses.push({
      id: (i + 1).toString(),
      date: formatDate(rows[i][0]),
      time: rows[i][1] instanceof Date ? Utilities.formatDate(rows[i][1], Session.getScriptTimeZone(), "HH:mm") : (rows[i][1] || "00:00"),
      desc: rows[i][2],
      category: rows[i][3],
      method: rows[i][4],
      amount: parseInt(rows[i][5]) || 0,
      status: rows[i][6] || "Pagado"
    });
  }

  return ContentService.createTextOutput(JSON.stringify(expenses))
    .setMimeType(ContentService.MimeType.JSON);
}

function formatDate(dateObj) {
  if (dateObj instanceof Date) {
    var yyyy = dateObj.getFullYear();
    var mm = String(dateObj.getMonth() + 1).padStart(2, '0');
    var dd = String(dateObj.getDate()).padStart(2, '0');
    return yyyy + '-' + mm + '-' + dd;
  }
  return dateObj;
}

function actualizarResumen() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var rows = ss.getSheets()[0].getDataRange().getValues();
  var meses = {}, semanas = {};
  for (var i = 1; i < rows.length; i++) {
    var f = String(formatDate(rows[i][0]));
    if (f.length < 10) continue;
    var ym = f.substring(0, 7), dia = parseInt(f.substring(8, 10), 10);
    var monto = parseInt(rows[i][5]) || 0;
    meses[ym] = meses[ym] || { n: 0, t: 0 };
    meses[ym].n++; meses[ym].t += monto;
    var w = semanaDe(ym, dia), k = ym + "|" + w.start;
    semanas[k] = semanas[k] || { ym: ym, w: w, n: 0, t: 0 };
    semanas[k].n++; semanas[k].t += monto;
  }
  var sh = ss.getSheetByName("Resumen") || ss.insertSheet("Resumen");
  sh.clear();
  var outM = Object.keys(meses).sort().reverse().map(function (k) { return [k, meses[k].n, meses[k].t]; });
  var outS = Object.keys(semanas).sort().reverse().map(function (k) {
    var s = semanas[k];
    return [s.ym, "Del " + s.w.start + " al " + s.w.end, s.n, s.t];
  });
  sh.getRange("A1:C1").setValues([["Mes", "Gastos", "Total"]]);
  sh.getRange("E1:H1").setValues([["Mes", "Semana", "Gastos", "Total"]]);
  if (outM.length) {
    sh.getRange(2, 1, outM.length, 1).setNumberFormat("@");
    sh.getRange(2, 1, outM.length, 3).setValues(outM);
    sh.getRange(2, 3, outM.length, 1).setNumberFormat("$#,##0");
  }
  if (outS.length) {
    sh.getRange(2, 5, outS.length, 2).setNumberFormat("@");
    sh.getRange(2, 5, outS.length, 4).setValues(outS);
    sh.getRange(2, 8, outS.length, 1).setNumberFormat("$#,##0");
  }
  sh.getRange("A1:H1").setFontWeight("bold");
  sh.autoResizeColumns(1, 8);
}

function semanaDe(ym, dia) {
  var y = parseInt(ym.substring(0, 4), 10), m = parseInt(ym.substring(5, 7), 10);
  var last = new Date(y, m, 0).getDate(), start = 1;
  while (start <= last) {
    var dow = (new Date(y, m - 1, start).getDay() + 6) % 7;
    var end = Math.min(last, start + (6 - dow));
    if (dia >= start && dia <= end) return { start: start, end: end };
    start = end + 1;
  }
}

Paso 2 · Gráficos automáticos en Sheets

Cada fila nueva recalcula fórmulas y gráficos sin que hagas nada.

1. Tabla dinámica

Selecciona tus filas ➔ Insertar ➔ Tabla dinámica. Pon Categoría en Filas y Monto en Valores (SUMA).

2. Gráfico circular o de columnas

Selecciona la tabla dinámica y elige Insertar ➔ Gráfico. Cada gasto nuevo actualiza totales y barras.

Paso 3 · Atajo de iPhone

Dos formas de enviar gastos a Sheets y a la app.

Opción A (recomendada): POST silencioso

En Atajos añade la acción "Obtener contenido de URL":
  • Método: POST
  • Encabezados: Content-Type = application/json
  • Cuerpo: JSON con amount, desc, method, category

Opción B (solo en Safari, no en la app instalada)

Añade la acción "Abrir URL" con este formato:
https://tu-app.com/index.html?amount=[Monto]&desc=[Comercio]&method=[Medio]&category=Supermercado
